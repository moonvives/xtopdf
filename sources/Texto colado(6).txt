        // --- P5.JS BACKGROUND ---
        function setup() {
            let canvas = createCanvas(2400, 3200);
            canvas.parent('p5-background');
            frameRate(30);
        }

        let particles = [];
        const numParticles = 120;

        function draw() {
            clear();
            stroke(0, 255, 157, 20);
            strokeWeight(1);

            if (particles.length === 0) {
                for (let i = 0; i < numParticles; i++) {
                    particles.push({
                        x: random(width),
                        y: random(height),
                        vx: random(-0.5, 0.5),
                        vy: random(-0.5, 0.5)
                    });
                }
            }

            for (let i = 0; i < particles.length; i++) {
                particles[i].x += particles[i].vx;
                particles[i].y += particles[i].vy;
                
                if (particles[i].x > width || particles[i].x < 0) particles[i].vx *= -1;
                if (particles[i].y > height || particles[i].y < 0) particles[i].vy *= -1;

                // Connect nearby particles
                for (let j = i + 1; j < particles.length; j++) {
                    let d = dist(particles[i].x, particles[i].y, particles[j].x, particles[j].y);
                    if (d < 180) {
                        stroke(0, 255, 157, map(d, 0, 180, 50, 0));
                        line(particles[i].x, particles[i].y, particles[j].x, particles[j].y);
                    }
                }
                
                noStroke();
                fill(0, 255, 157, 150);
                circle(particles[i].x, particles[i].y, 3);
            }
        }
