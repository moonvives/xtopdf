        // --- P5.JS BACKGROUND ---
        function setup() {
            let canvas = createCanvas(2400, 3200);
            canvas.parent('p5-background');
            frameRate(30);
        }

        let particles = [];
        const numParticles = 100;

        function draw() {
            clear();
            stroke(255, 255, 255, 30);
            strokeWeight(2);

            if (particles.length === 0) {
                for (let i = 0; i < numParticles; i++) {
                    particles.push(createVector(random(width), random(height)));
                }
            }

            for (let i = 0; i < particles.length; i++) {
                particles[i].x += random(-1, 1);
                particles[i].y += random(-1, 1);
                
                if (particles[i].x > width) particles[i].x = 0;
                if (particles[i].x < 0) particles[i].x = width;
                if (particles[i].y > height) particles[i].y = 0;
                if (particles[i].y < 0) particles[i].y = height;

                for (let j = i + 1; j < particles.length; j++) {
                    let d = dist(particles[i].x, particles[i].y, particles[j].x, particles[j].y);
                    if (d < 150) {
                        stroke(0, 255, 157, map(d, 0, 150, 100, 0));
                        line(particles[i].x, particles[i].y, particles[j].x, particles[j].y);
                    }
                }
                
                noStroke();
                fill(0, 255, 157, 150);
                circle(particles[i].x, particles[i].y, 4);
            }
        }
